# API Documentation

Base URL: http://localhost:5000


-----------------------------------------------------------------------------------------
-----------------------------------------------------------------------------------------


1. authentication endpoints

# register seller
  url               : /api/register/seller
  method            : POST
  description       : registers a new seller account
  expected json body:
  {
    "fullName": "seller1",
    "email": "seller1@gmail.com",
    "password": "password",
    "businessAddress": "house"
  }

* register buyer
  url               : /api/register/buyer
  method            : POST
  description       : registers a new buyer account
  expected json body:
  {
    "fullName": "buyer1",
    "email": "buyer1@gmail.com",
    "password": "password"
  }


# login seller
  url               : /api/login/seller
  method            : POST
  description       : logs seller in and creates a session
  expected json body:
  {
    "email": "seller1@gmail.com",
    "password": "password"
  }


# login buyer
  url               : /api/login/buyer
  method            : POST
  description       : logs buyer in and creates a session
  expected json body:
  {
    "email": "buyer1@gmail.com",
    "password": "password"
  }

# logout
  url               : /api/logout
  method            : POST
  description       : logs off user and invalidates the cookie
  expected json body: none

2. product endpoints

# get all products
  url               : /api/products
  method            : GET
  description       : returns all products in json format
  expected json body: none


# get product by id
  url               : /api/products/:id
  method            : GET
  description       : returns exactly one product in json
  expected json body: none



# create product
  url               : /api/products/create
  method            : POST
  description       : creates new product. Requires seller authentication
  expected json body:
  {
    "title": "Wireless Mouse",
    "description": "Ergonomic wireless",
    "price": 29.99,
    "stock": 150
  }

3. cart endpoints

# add product to cart
  url               : /api/cart/items
  method            : POST
  description       : creates cart if not exists. adds product units to cart_item
  expected json body:
  {
    "productId": 3,
    "quantity": 10
  }

# get cart items
  url               : /api/cart
  method            : GET
  description       : returns all items in the cart for the logged-in buyer
  expected json body: none

# delete cart item
  url               : /api/cart/items/:unitId
  method            : DELETE
  description       : deletes a specific item from the cart for the logged-in buyer
  expected json body: none

4. Placing an order

# place an order
  url		    : /api/orders
  method 	    : POST
  description       : places the order, deletes the cart, assigns any free delivery boy to that order automatically
  expected json body: (the address ID must belong to that specific buyer, it is checked whether or not the address belongs to the   current buyer or not)
  {
    "addressId_": 2, 
    "receiverName": "Mijo",
    "receiverPhoneNumber": "01850286110"
  }

if a delivery boy is open, assign him to the order [order.status -> 'shipped', delivery_boy.status -> 'assigned']
else [order.status -> 'pending']

# order has been delivered and completed
  url		    : /admin/orders/:orderId/complete
  method 	    : POST
  description       : admin marks an order as completed, then the delivery boy who was assigned to that order will be available again. If there is a pending order, it will be assigned to that recently available delivery boy.
  expected json body: none

5. review endpoints

# get product reviews
  url               : /api/product/:productId/reviews
  method            : GET
  description       : returns all reviews and ratings for a specific product in json format
  expected json body: none

# post product review
  url               : /api/product/:productId/review
  method            : POST
  description       : creates a review for a product. Requires buyer authentication and that the buyer has purchased and received the product.
  expected json body:
  {
    "rating": 5,
    "reviewText": "Great product, very comfortable!"
  }


# edit product review
  url               : /api/product/:productId/review
  method            : PUT
  description       : updates an existing review for a product. Requires buyer authentication.
  expected json body:
  {
    "rating": 4,
    "reviewText": "Updated review text!"
  }

# delete product review
  url               : /api/product/:productId/review
  method            : DELETE
  description       : deletes the logged-in buyer's review for a product. Requires buyer authentication.
  expected json body: none

6. address endpoints

# add address
  url               : /api/address
  method            : POST
  description       : adds a new shipping address for the logged-in buyer
  expected json body:
  {
    "label": "Home",
    "street": "123 Main St",
    "city": "New York",
    "postalCode": "10001",
    "country": "USA",
    "isCurrent": true
  }

# get all addresses
  url               : /api/addresses
  method            : GET
  description       : returns all saved addresses for the logged-in buyer
  expected json body: none

# get current address
  url               : /api/address/current
  method            : GET
  description       : returns the currently selected default shipping address for the logged-in buyer
  expected json body: none

# set current address
  url               : /api/address/:addressId/current
  method            : PUT
  description       : sets a specific saved address as the current active shipping address
  expected json body: none

# delete address
  url               : /api/address/:addressId
  method            : DELETE
  description       : deletes a specific address (fails if linked to past orders)
  expected json body: none
