# API Documentation

Base URL: http://localhost:5000


-----------------------------------------------------------------------------------------
-----------------------------------------------------------------------------------------


1. authentication endpoints

# register seller
  url               : /api/register/seller
  method            : POST
  description       : registers a new seller account
  expected json body:
  {
    "fullName": "seller1",
    "email": "seller1@gmail.com",
    "password": "password",
    "businessAddress": "house"
  }

* register buyer
  url               : /api/register/buyer
  method            : POST
  description       : registers a new buyer account
  expected json body:
  {
    "fullName": "buyer1",
    "email": "buyer1@gmail.com",
    "password": "password"
  }


# login seller
  url               : /api/login/seller
  method            : POST
  description       : logs seller in and creates a session
  expected json body:
  {
    "email": "seller1@gmail.com",
    "password": "password"
  }


# login buyer
  url               : /api/login/buyer
  method            : POST
  description       : logs buyer in and creates a session
  expected json body:
  {
    "email": "buyer1@gmail.com",
    "password": "password"
  }


# login admin
  url               : /api/login/admin
  method            : POST
  description       : logs admin in and creates a session. Required for category create/update/delete and completing orders
  expected json body:
  {
    "email": "admin@gmail.com",
    "password": "password"
  }

# get current session
  url               : /api/session
  method            : GET
  description       : returns the current session. role is "buyer", "seller", or "admin" when logged in
  expected json body: none
  example response (logged in):
  {
    "authenticated": true,
    "role": "buyer",
    "id": 1
  }
  example response (logged out):
  {
    "authenticated": false
  }

# logout
  url               : /api/logout
  method            : POST
  description       : logs off user and invalidates the cookie
  expected json body: none

2. product endpoints

# get all products
  url               : /api/products
  method            : GET
  description       : returns all products in json format
  expected json body: none


# get product by id
  url               : /api/products/:id
  method            : GET
  description       : returns exactly one product in json (includes images and categories)
  expected json body: none


# get products by category
  url               : /api/products/category/:categoryId
  method            : GET
  description       : returns all products assigned to that category. Returns 404 if the category does not exist
  expected json body: none


# get seller products
  url               : /api/seller/products
  method            : GET
  description       : returns all products belonging to the logged-in seller (includes images and categories). Requires seller authentication
  expected json body: none


# create product
  url               : /api/products/create
  method            : POST
  description       : creates new product. Requires seller authentication. Optional images can be uploaded in the same request (first image becomes primary)
  expected json body:
  {
    "name": "Wireless Mouse",
    "description": "Ergonomic wireless",
    "price": 29.99,
    "stock": 150
  }


# update product
  url               : /api/products/:productId
  method            : PUT
  description       : updates a product owned by the logged-in seller. additionalStock adds that many new units. Optional new images can be uploaded. Requires seller authentication
  expected json body:
  {
    "name": "Wireless Mouse"
    "description": "Updated description"
    "price": 24.99
    "additionalStock": 10
    "images": [file1, file2, ...] (optional)
  }


# adding a product to a category
  url               : /api/products/:productId/categories/:categoryId
  method            : POST
  description       : assigns a product to a category. Requires seller authentication. Product must belong to the logged-in seller
  expected json body: none


# remove a product from a category
  url               : /api/products/:productId/categories/:categoryId
  method            : DELETE
  description       : removes a product from a category. Requires seller authentication. Product must belong to the logged-in seller
  expected json body: none

4. Placing an order

# place an order
  url		            : /api/orders
  method 	          : POST
  description       : places the order, deletes the cart, assigns any free delivery boy to that order automatically
  expected json body: (the address ID must belong to that specific buyer, it is checked whether or not the address belongs to the   current buyer or not)
  {
    "addressId_": 2, 
    "receiverName": "Mijo",
    "receiverPhoneNumber": "01850286110"
  }

 if a delivery boy is open, assign him to the order [order.status -> 'shipped', delivery_boy.status -> 'assigned']
 else [order.status -> 'pending']

# order has been delivered and completed
  url		            : /api/admin/orders/:orderId/complete
  method 	          : POST
  description       : admin marks an order as completed, then the delivery boy who was assigned to that order will be available again. If there is a pending order, it will be assigned to that recently available delivery boy. Requires admin authentication.
  expected json body: none

# get buyer orders
  url               : /api/orders
  method            : GET
  description       : returns all past and active orders for the logged-in buyer with items and images. Requires buyer authentication.
  expected json body: none

# get one buyer order
  url               : /api/orders/:orderId
  method            : GET
  description       : returns one order belonging to the logged-in buyer, with items. Requires buyer authentication.
  expected json body: none

5. review endpoints

# get product reviews
  url               : /api/product/:productId/reviews
  method            : GET
  description       : returns all reviews and ratings for a specific product in json format
  expected json body: none

# post product review
  url               : /api/product/:productId/review
  method            : POST
  description       : creates a review for a product. Requires buyer authentication and that the buyer has purchased and received the product.
  expected json body:
  {
    "rating": 5,
    "reviewText": "Great product, very comfortable!"
  }


# edit product review
  url               : /api/product/:productId/review
  method            : PUT
  description       : updates an existing review for a product. Requires buyer authentication.
  expected json body:
  {
    "rating": 4,
    "reviewText": "Updated review text!"
  }

# delete product review
  url               : /api/product/:productId/review
  method            : DELETE
  description       : deletes the logged-in buyer's review for a product. Requires buyer authentication.
  expected json body: none

6. address endpoints

# add address
  url               : /api/address
  method            : POST
  description       : adds a new shipping address for the logged-in buyer
  expected json body:
  {
    "label": "Home",
    "street": "123 Main St",
    "city": "New York",
    "postalCode": "10001",
    "country": "USA",
    "isCurrent": true
  }

# get all addresses
  url               : /api/addresses
  method            : GET
  description       : returns all saved addresses for the logged-in buyer
  expected json body: none

# get current address
  url               : /api/address/current
  method            : GET
  description       : returns the currently selected default shipping address for the logged-in buyer
  expected json body: none

# set current address
  url               : /api/address/:addressId/current
  method            : PUT
  description       : sets a specific saved address as the current active shipping address
  expected json body: none

# delete address
  url               : /api/address/:addressId
  method            : DELETE
  description       : deletes a specific address (fails if linked to past orders)
  expected json body: none

  method            : POST

    - images: [file1, file2, ...]

  description       : sets a specific image as the primary image for a product. Requires seller authentication.
  expected json body: none

# delete product image
  url               : /api/images/:imageId
  method            : DELETE
  description       : deletes an image from the database and the server disk. Requires seller authentication.
  expected json body: none

8. Category

# add new Category
  url               : /api/categories
  method            : POST
  description       : Creates a new Category. Requires admin authentication
  expected json body: 
  {
    "categoryName": "Something",
    "parentCategoryId": 1
  }
  parentCategoryId is optional. Omit it if this category is a top-level parent.

# get all categories
  url               : /api/categories
  method            : GET
  description       : shows all categories. Requires login (buyer, seller, or admin)
  expected json body: none

# get 1 category
  url               : /api/categories/:categoryId
  method            : GET
  description       : shows that specific category. Requires login (buyer, seller, or admin)
  expected json body: none

# update a category's parent and name
  url               : /api/categories/:categoryId
  method            : PUT
  description       : updates that category's name and parent. Requires admin authentication
  expected json body:
  {
    "categoryName": "Smartphones",
    "parentCategoryId": 1
  }

# delete a category
  url               : /api/categories/:categoryId
  method            : DELETE
  description       : deletes that specified category. Fails if it still has child categories. Product links to this category are removed. Requires admin authentication
  expected json body: none

9. Discount

#create discount (logged in as admin)
url               : /api/discounts
method            : POST
description       : creates a new discount
expected json body:
{
  "discountName": "Summer Sale",
  "discountType": "percentage",
  "discountValue": 20,
  "startDate": "2026-09-24T00:00:00Z",
  "endDate": "2026-10-01T00:00:00Z"
}

# get all discounts
url               : /api/discounts
method            : GET
description       : shows all discounts
expected json body: none

# get discount by ID
url               : /api/discounts/:discountId
method            : GET
description       : shows all discounts
expected json body: none

# update discount
url               : /api/discounts/:discountId
method            : PUT
description       : updates information of discount
expected json body: (if a one or more parameters are missing, then the endpoint will use the old value of the existing discount)
{
  "discountName": "Updated Summer Sale",
  "discountType": "percentage",
  "discountValue": 20,
  "startDate": "2026-09-24T00:00:00Z",
  "endDate": "2026-10-01T00:00:00Z"
}
